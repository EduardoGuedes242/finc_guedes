CREATE TABLE CATEGORIAS (
  ID     integer PRIMARY KEY AUTOINCREMENT,
  NOME   text NOT NULL,
  TIPO   char(1) NOT NULL,
  COR    text,
  ICONE  text,
  ATIVO  integer NOT NULL DEFAULT 1,
  /* Checks */
  CHECK (TIPO IN ('E','S'))
);

CREATE INDEX IDX_CATEGORIAS_TIPO
  ON CATEGORIAS
  (TIPO);

CREATE TABLE MOVIMENTOS (
  ID               integer PRIMARY KEY AUTOINCREMENT,
  RECORRENCIA_ID   integer,
  CATEGORIA_ID     integer NOT NULL,
  TIPO             char(1) NOT NULL,
  DESCRICAO        text NOT NULL,
  DATA_VENCIMENTO  text NOT NULL,
  DATA_PAGAMENTO   text,
  VALOR_PREVISTO   integer NOT NULL,
  VALOR_PAGO       integer,
  STATUS           integer NOT NULL DEFAULT 0,
  ORIGEM           integer NOT NULL DEFAULT 0,
  OBSERVACAO       text,
  CRIADO_EM        text NOT NULL DEFAULT CURRENT_TIMESTAMP,
  ALTERADO_EM      text,
  /* Checks */
  CHECK (TIPO IN ('E','S')),
  /* Foreign keys */
  FOREIGN KEY (RECORRENCIA_ID)
    REFERENCES RECORRENCIAS(ID)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION, 
  FOREIGN KEY (CATEGORIA_ID)
    REFERENCES CATEGORIAS(ID)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);

CREATE INDEX IDX_MOVIMENTOS_CATEGORIA
  ON MOVIMENTOS
  (CATEGORIA_ID);

CREATE INDEX IDX_MOVIMENTOS_DATA
  ON MOVIMENTOS
  (DATA_VENCIMENTO);

CREATE INDEX IDX_MOVIMENTOS_RECORRENCIA
  ON MOVIMENTOS
  (RECORRENCIA_ID);

CREATE INDEX IDX_MOVIMENTOS_STATUS
  ON MOVIMENTOS
  (STATUS);

CREATE TABLE RECORRENCIAS (
  ID                         integer PRIMARY KEY AUTOINCREMENT,
  CATEGORIA_ID               integer NOT NULL,
  TIPO                       char(1) NOT NULL,
  DESCRICAO                  text NOT NULL,
  VALOR_PREVISTO             integer NOT NULL,
  DIA_VENCIMENTO             integer NOT NULL,
  DATA_INICIO                text NOT NULL,
  DATA_FIM                   text,
  ATIVO                      integer NOT NULL DEFAULT 1,
  GERAR_ANTECIPADO           integer NOT NULL DEFAULT 0,
  ULTIMA_COMPETENCIA_GERADA  text,
  OBSERVACAO                 text,
  /* Checks */
  CHECK (TIPO IN ('E','S')), 
  CHECK (DIA_VENCIMENTO BETWEEN 1 AND 31),
  /* Foreign keys */
  FOREIGN KEY (CATEGORIA_ID)
    REFERENCES CATEGORIAS(ID)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION
);

CREATE INDEX IDX_RECORRENCIAS_ATIVO
  ON RECORRENCIAS
  (ATIVO);
